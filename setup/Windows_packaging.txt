# Windows jpackage Command
# -------------------------------------------------------------
# Run this command from the root directory of the project.

jpackage ^
  --type exe ^
  --name Napster ^
  --app-version 1.0.0 ^
  --input .\input ^
  --main-jar Napster.jar ^
  --main-class Napster.App ^
  --module-path .\lib\javafx-jmods-26.0.1 ^
  --add-modules javafx.controls,javafx.fxml,javafx.media,java.sql,java.net.http,java.desktop ^
  --java-options "-Djava.library.path=$APPDIR" ^
  --java-options "-Dapp.dir=$APPDIR" ^
  --java-options "-Dprism.order=d3d" ^
  --java-options "--enable-native-access=javafx.graphics,javafx.media,ALL-UNNAMED" ^
  --icon .\packaging\icon.ico ^
  --dest .\dist

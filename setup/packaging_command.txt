// type -> pkg for MacOS, exe for Windows
// module-path seperator -> : for MacOS, ; for Windows

jpackage \
  --type dmg \
  --name Napster \
  --app-version 1.0.0 \
  --input ./input \
  --main-jar Napster.jar \
  --main-class Napster.App \
  --module-path ./lib/javafx-sdk-26.0.1/lib \
  --add-modules javafx.controls,javafx.fxml,javafx.media \
  --java-options "--enable-native-access=javafx.graphics,javafx.media,ALL-UNNAMED" \
  --icon ./packaging/icon.icns \
  --dest ./dist
